option(MIXIMUS_ENABLE_CUDA "Build optional CUDA/Vulkan transfers when the toolkit is available" ON)
if(NOT MIXIMUS_ENABLE_CUDA)
    return()
endif()
find_package(CUDAToolkit QUIET)
if(CUDAToolkit_FOUND AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(cuda_dependencies INTERFACE)
    target_link_libraries(cuda_dependencies INTERFACE CUDA::cudart)
    message(STATUS "CUDA/Vulkan transfers enabled (${CUDAToolkit_VERSION}); selected automatically when available; force staging with --disable-cuda")
else()
    message(STATUS "CUDA/Vulkan transfers unavailable on this build; using Vulkan staging")
endif()
